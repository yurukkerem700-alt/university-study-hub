1. Soru: Proje adı ve temel motivasyonu ve özgünlük değerlendirmelerinizi yazınız. (maksimum 200 kelime)

Proje adı: Notora – Üniversite Çalışma Merkezi.

Temel motivasyon: Üniversite öğrencisi ders notunu bir yerde, sınav ve burs tarihini başka yerde, güvenilir kaynağı üçüncü bir yerde arıyor. Notlar mesaj gruplarında kayboluyor, kaynağın doğruluğu belirsiz kalıyor, çalışırken önemli yerleri işaretleyecek düzenli bir ortam yok. Notora bu dağınıklığı tek web sitesinde topluyor: öğrenciler not yükleyip birbirine puan veriyor; 51 resmî ve akademik kaynak türüne göre ayrılmış listeleniyor; 2026 sınav ve burs takvimi ile 90 başlıklı ders haritası sunuluyor; GANO, Pomodoro ve günlük görev araçları var. Çalışma Defteri bölümünde öğrenci kendi notunu yazıp okurken önemli yerleri renkle işaretliyor, işaretten bilgi kartı üretip aralıklı tekrarla çalışıyor.

Özgünlük değerlendirmesi: Benzer siteler genellikle yalnızca not paylaşıyor ya da yalnızca takvim veriyor. Notora kaynağı üç güven sınıfına ayırıyor (resmî, doğrulanmış dış, öğrenci) ve dış içeriği kopyalamadan bağlantı veriyor. 224 üniversite ve 21.542 program kaydı YÖK Atlas açık verisinden alındı. Veriler Supabase (PostgreSQL) üzerinde, satır düzeyinde güvenlikle (RLS) saklanıyor; herkes yalnızca kendi verisini değiştirebiliyor. Site PWA olduğundan telefona kurulup çevrimdışı açılabiliyor.

5. Soru: Projeye ait Veritabanı ilişkisel yapısını ve tablolara ait SQL oluşturma kodlarını yazınız.
İlişkisel yapı (15 tablo; PostgreSQL / Supabase)
auth.users 1—1 profilesid = users.id
auth.users 1—1 user_preferencesid = users.id
auth.users 1—N notesuploader_id
notes N—M users (beğeni)note_likes(note_id,user_id)
notes N—M users (kaydetme)bookmarks(note_id,user_id)
notes N—M users (puan)note_ratings(note_id,user_id)
notes 1—N note_download_eventsnote_id
users 1—N study_notebooksuser_id
study_notebooks 1—N study_highlightsnotebook_id
study_notebooks 1—N study_flashcardsnotebook_id
create table profiles (
  id uuid primary key references auth.users(id), display_name text
  not null default 'Öğrenci', university text, department text,
  study_year smallint check (study_year between 1 and 8), role
  text not null default 'student' check (role in
  ('student','teacher','admin')), created_at timestamptz now(),
  updated_at timestamptz now()
);
create table user_preferences (
  id uuid primary key references auth.users(id), education_level
  text default 'Lisans', university text, department text,
  study_year smallint, goal text, daily_minutes smallint default
  60, focus_mode text default 'pomodoro', theme text default
  'system', notifications boolean default true, preferred_topics
  text[] default '{}', updated_at timestamptz now()
);
create table notes (
  id bigint identity primary key, title text not null, description
  text, department text not null, study_year smallint check
  (study_year between 1 and 8), topic text, content_type text not
  null default 'Ders Notu', university text, lecturer text,
  file_path text, file_name text, file_size bigint, downloads
  integer not null default 0 check (downloads >= 0), likes_count
  integer not null default 0 check (likes_count >= 0), rating
  numeric not null default 0 check (rating between 0 and 5),
  ratings_count integer not null default 0 check (ratings_count >=
  0), uploader_id uuid references auth.users(id), status text not
  null default 'published' check (status in
  ('pending','published','rejected')), created_at timestamptz
  now(), updated_at timestamptz now()
);
create table note_likes (
  note_id bigint not null references notes(id), user_id uuid not
  null references auth.users(id), created_at timestamptz now(),
  primary key (note_id, user_id)
);
create table bookmarks (
  note_id bigint not null references notes(id), user_id uuid not
  null references auth.users(id), created_at timestamptz now(),
  primary key (note_id, user_id)
);
create table note_ratings (
  note_id bigint not null references notes(id), user_id uuid not
  null references auth.users(id), rating smallint not null check
  (rating between 1 and 5), created_at timestamptz now(),
  updated_at timestamptz now(), primary key (note_id, user_id)
);
create table note_download_events (
  id bigint identity primary key, note_id bigint not null
  references notes(id), user_id uuid references auth.users(id),
  client_token text not null, created_at timestamptz now(), unique
  (note_id, client_token)
);
create table knowledge_resources (
  id bigint identity primary key, title text not null, summary
  text not null, category text not null, resource_type text not
  null default 'web', tags text[] not null default '{}',
  source_name text not null, source_url text not null, authority
  text not null default 'official', verified_at date not null
  default current_date, last_checked_at timestamptz now(),
  featured boolean not null default false, created_at timestamptz
  now(), domain text, year_level smallint check (year_level
  between 1 and 4), subject text, topic text
);
create table academic_events (
  id bigint identity primary key, title text not null, organizer
  text not null, event_date timestamptz not null, deadline_date
  timestamptz, category text not null, source_url text not null,
  source_name text not null, created_at timestamptz now()
);
create table study_catalog (
  id bigint identity primary key, domain text not null, year_level
  smallint not null check (year_level between 1 and 4), semester
  smallint check (semester between 1 and 2), subject text not
  null, topic text not null, skill text, level text not null
  default 'temel' check (level in ('temel','orta','ileri')),
  roadmap_type text not null default 'genel' check (roadmap_type
  in ('genel','alan')), created_at timestamptz now()
);
create table universities_catalog (
  id bigint identity primary key, name text not null unique,
  university_type text not null default 'Devlet', city text,
  country_region text not null default 'Türkiye', source_year
  smallint not null default 2025, source_name text not null
  default 'YÖK Atlas 2025 tabanlı açık veri'
);
create table programs_catalog (
  id bigint identity primary key, name text not null, level text
  not null, score_type text, duration_years text, faculty_examples
  text[] not null default '{}', source_year smallint not null
  default 2025, source_name text not null default 'YÖK Atlas 2025
  tabanlı açık veri', unique (name, level, score_type,
  duration_years)
);
create table study_notebooks (
  id bigint identity primary key, user_id uuid not null default
  auth.uid() references auth.users(id) on delete cascade, title
  text not null check (char_length(title) between 1 and 120),
  course text, body text not null default '', created_at
  timestamptz now(), updated_at timestamptz now()
);
create table study_highlights (
  id bigint identity primary key, notebook_id bigint not null
  references study_notebooks(id) on delete cascade, user_id uuid
  not null default auth.uid() references auth.users(id) on delete
  cascade, start_pos integer not null check (start_pos >= 0),
  end_pos integer not null, color text not null default 'yellow'
  check (color in ('yellow','green','blue','pink')), important
  boolean not null default false, comment text, created_at
  timestamptz now(), check (end_pos > start_pos)
);
create table study_flashcards (
  id bigint identity primary key, notebook_id bigint references
  study_notebooks(id) on delete cascade, user_id uuid not null
  default auth.uid() references auth.users(id) on delete cascade,
  front text not null, back text not null, box smallint not null
  default 1 check (box between 1 and 5), next_review date not null
  default current_date, created_at timestamptz now()
);
create policy study_notebooks_own on study_notebooks for all to
  authenticated using (user_id = (select auth.uid())) with check
  (user_id = (select auth.uid()));
create policy study_highlights_own on study_highlights for all to
  authenticated using (user_id = (select auth.uid())) with check
  (user_id = (select auth.uid()));
create policy study_flashcards_own on study_flashcards for all to
  authenticated using (user_id = (select auth.uid())) with check
  (user_id = (select auth.uid()));
